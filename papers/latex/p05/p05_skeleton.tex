\begin{abstract}
We study the contractive properties of the map family
$\Phi_{\alpha,\ell}(\Psi) = r_{\alpha,\ell} R_{\alpha,\ell} \Psi \tilde{R}_{\alpha,\ell} + \Delta_{\alpha,\ell}(\Psi)$
in the CT runtime. Empirical evidence shows individual contractivity ($c < 1$)
for all tested maps. Hutchinson convergence confirmed for conservative
configurations ($r + \mathrm{Lip}(\Delta) < 0.85$). Fixed-point uniqueness
confirmed from 20 random starts. Global uniform proof left for future work.
\end{abstract}

\section{Formal Setting}
State space: $X = \bigoplus_{\ell} C(\Omega, \mathrm{Cl}_{3,0}^{(\ell)})$ with sup norm.

\begin{theorem}[Banach -- Variant A]
If $L_\ell = \sum |w_{\alpha,\ell}|(r_{\alpha,\ell} + L_{\alpha,\ell}) < 1$,
then $T_\ell$ has a unique fixed point $A^*_\ell$.
\end{theorem}

\begin{theorem}[Hutchinson '81 -- Variant B]
If every $\Phi_{\alpha,\ell}$ satisfies $c_{\alpha,\ell} < 1$, then
$H_\ell(K) = \bigcup_\alpha \Phi_{\alpha,\ell}(K)$ has unique compact
invariant $K^*_\ell$ in Hausdorff metric.
\end{theorem}

Contractivity margin: $m_{\alpha,\ell} := 1 - c_{\alpha,\ell}$.

\section{Rotor Isometry}
\begin{claim}[1: Pure rotor preserves norm]
$R\Psi\tilde{R}$ with $R\tilde{R}=1$ gives contraction ratio $= 1.0$ exactly.
Isometry, not contraction. Must be damped for Hutchinson.
\end{claim}
\noindent\textbf{Test}: \texttt{TestRotorIsometry} (2 tests, 50 runs).\\[4pt]

\section{Limitations}
\begin{itemize}
  \item Contractivity is sampled evidence, NOT a global uniform proof.
  \item gamma\_flow and defect\_stab marginally contractive ($m \approx 10^{-3}$).
  \item Pure rotors ($c = 1$) excluded from strong Hutchinson unless damped.
  \item Admissibility preservation not proven.
  \item Fixed-point representative is NOT a barycenter without invariant measure.
\end{itemize}
